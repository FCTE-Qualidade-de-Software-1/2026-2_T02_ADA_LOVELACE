  55:     await entries(filesStore).then((entries) => {
  56:       for (const [id, imageData] of entries as [FileId, BinaryFileData][]) {
  57:         // if image is unused (not on canvas) & is older than 1 day, delete it

  76: ) => {
  77:   const localStorageQuotaExceeded = appJotaiStore.get(

 122: type SavingLockTypes = "collaboration";

 124: export class LocalData {
 125:   private static _save = debounce(
 126:     async (
 127:       elements: readonly ExcalidrawElement[],
 128:       appState: AppState,
 129:       files: BinaryFiles,
 130:       onFilesSaved: () => void,
 131:     ) => {
 132:       saveDataStateToLocalStorage(elements, appState);
 133:
 134:       await this.fileStorage.saveFiles({
 135:         elements,
 136:         files,
 137:       });
 138:       onFilesSaved();
 139:     },
 140:     SAVE_TO_LOCAL_STORAGE_TIMEOUT,
 141:   );
 142:
 143:   /** Saves DataState, including files. Bails if saving is paused */
 144:   static save = (
 145:     elements: readonly ExcalidrawElement[],
 146:     appState: AppState,
 147:     files: BinaryFiles,
 148:     onFilesSaved: () => void,
 149:   ) => {
 150:     // we need to make the `isSavePaused` check synchronously (undebounced)
 151:     if (!this.isSavePaused()) {
 152:       this._save(elements, appState, files, onFilesSaved);
 153:     }
 154:   };
 155:
 156:   static flushSave = () => {
 157:     this._save.flush();
 158:   };
 159:
 160:   private static locker = new Locker<SavingLockTypes>();
 161:
 162:   static pauseSave = (lockType: SavingLockTypes) => {
 163:     this.locker.lock(lockType);
 164:   };
 165:
 166:   static resumeSave = (lockType: SavingLockTypes) => {
 167:     this.locker.unlock(lockType);
 168:   };
 169:
 170:   static isSavePaused = () => {
 171:     return document.hidden || this.locker.isLocked();
 172:   };
 173:
 174:   // ---------------------------------------------------------------------------
 175:
 176:   static fileStorage = new LocalFileManager({
 177:     onFileStatusChange: FileStatusStore.updateStatuses.bind(FileStatusStore),
 178:     getFiles(ids) {
 179:       return getMany(ids, filesStore).then(
 180:         async (filesData: (BinaryFileData | undefined)[]) => {
 181:           const loadedFiles: BinaryFileData[] = [];
 182:           const erroredFiles = new Map<FileId, true>();
 183:
 184:           const filesToSave: [FileId, BinaryFileData][] = [];
 185:
 186:           filesData.forEach((data, index) => {
 187:             const id = ids[index];
 188:             if (data) {
 189:               const _data: BinaryFileData = {
 190:                 ...data,
 191:                 lastRetrieved: Date.now(),
 192:               };
 193:               filesToSave.push([id, _data]);
 194:               loadedFiles.push(_data);
 195:             } else {
 196:               erroredFiles.set(id, true);
 197:             }
 198:           });
 199:
 200:           try {
 201:             // save loaded files back to storage with updated `lastRetrieved`
 202:             setMany(filesToSave, filesStore);
 203:           } catch (error) {
 204:             console.warn(error);
 205:           }
 206:
 207:           return { loadedFiles, erroredFiles };
 208:         },
 209:       );
 210:     },
 211:     async saveFiles({ addedFiles }) {
 212:       const savedFiles = new Map<FileId, BinaryFileData>();
 213:       const erroredFiles = new Map<FileId, BinaryFileData>();
 214:
 215:       // before we use `storage` event synchronization, let's update the flag
 216:       // optimistically. Hopefully nothing fails, and an IDB read executed
 217:       // before an IDB write finishes will read the latest value.
 218:       updateBrowserStateVersion(STORAGE_KEYS.VERSION_FILES);
 219:
 220:       await Promise.all(
 221:         [...addedFiles].map(async ([id, fileData]) => {
 222:           try {
 223:             await set(id, fileData, filesStore);
 224:             savedFiles.set(id, fileData);
 225:           } catch (error: any) {
 226:             console.error(error);
 227:             erroredFiles.set(id, fileData);
 228:           }
 229:         }),
 230:       );
 231:
 232:       return { savedFiles, erroredFiles };
 233:     },
 234:   });
